**Patron de Resolucion de Menu:**
```typescript
// Logica de prioridad:
// 1. Buscar DailyMenuAssignment para la fecha
// 2. Si no existe, buscar WeeklyPattern para el dia de la semana
// 3. Si no hay patron, usar MenuItem.availableDays (sistema legacy)
const assignment = await prisma.dailyMenuAssignment.findUnique({
  where: { cafeteriaId_date: { cafeteriaId, date } }
});
if (assignment) return { source: 'assignment', items };

const pattern = await prisma.weeklyPattern.findUnique({ where: { cafeteriaId } });
if (pattern) return { source: 'pattern', items };

return { source: 'default', items: await getDefaultMenu(dayOfWeek) };
```

  const today = new Date();

### US-203: Creacion de Tickets con Precio

**Archivo:** `packages/api/src/routes/payments.routes.ts`

**Logica Implementada:**
```typescript
// Calcular precio por ticket
const pricePerTicket = Math.floor(rechargePackage.price / rechargePackage.ticketCount);

// Si ya tiene tickets, calcular promedio ponderado
if (existingTicket) {
  const totalExistingValue = existingTicket.quantity * existingTicket.pricePerTicket;
  const totalNewValue = rechargePackage.ticketCount * pricePerTicket;
  const totalQuantity = existingTicket.quantity + rechargePackage.ticketCount;
  const newPricePerTicket = Math.floor((totalExistingValue + totalNewValue) / totalQuantity);

  await tx.studentTicket.update({
    where: { id: existingTicket.id },
    data: { quantity: totalQuantity, pricePerTicket: newPricePerTicket }
  });
} else {
  await tx.studentTicket.create({
    data: { studentId, ticketType, quantity: rechargePackage.ticketCount, pricePerTicket }
  });
}
```
