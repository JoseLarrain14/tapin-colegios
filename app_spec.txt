<project_specification>
  <project_name>Tap In Colegios</project_name>

  <overview>
    Tap In Colegios es una plataforma que permite a apoderados (padres) cargar saldo digital o comprar paquetes de tickets para que sus hijos puedan retirar comida en el casino escolar. El sistema soporta multiples modelos de negocio: solo tickets, solo saldo libre (kiosko), o modelo mixto, configurable por cada colegio. Los estudiantes validan su identidad mediante huella dactilar o busqueda por RUT al momento de retirar.
  </overview>

  <target_audience>
    <primary>Apoderados/Padres de estudiantes en Chile</primary>

    <tertiary>Administradores de colegios</tertiary>
  </target_audience>

      <wallets>
        - id (UUID, PK)
        - student_id (UUID, FK students, UNIQUE)

        - expires_at (TIMESTAMPTZ, opcional)

        - id (UUID, PK)

        - created_at (TIMESTAMPTZ)
      </wallet_logs>

      <transactions>
        - id (UUID, PK)
        - wallet_id (UUID, FK wallets)

        - amount (INTEGER)
        - description (TEXT)
        - items (JSONB) - [{item_id, name, price, quantity}]
        - tickets_used (JSONB) - [{ticket_type, quantity}]
        - validated_by (UUID, FK users) - operador
        - validation_method (ENUM: fingerprint, rut_search, qr_code, app_order)
        - source (ENUM: app, casino) - donde se origino
        - created_at (TIMESTAMPTZ)
      </transactions>

      <orders>
        - id (UUID, PK)
        - guardian_id (UUID, FK guardians)
        - student_id (UUID, FK students)
        - cafeteria_id (UUID, FK cafeterias)
        - status (ENUM: pending, confirmed, preparing, ready, delivered, cancelled)

        - pickup_time (TIME)
        - items (JSONB)
        - total (INTEGER)
        - comments (TEXT)
        - created_at, updated_at (TIMESTAMPTZ)
      </orders>

      <payments>
        - id (UUID, PK)
        - guardian_id (UUID, FK guardians)
        - student_id (UUID, FK students)
        - wallet_id (UUID, FK wallets)
        - recharge_package_id (UUID, FK recharge_packages, nullable)
        - amount (INTEGER)
        - gateway (ENUM: webpay, flow, fintoc, mock)
        - gateway_tx_id (VARCHAR)
        - status (ENUM: pending, processing, completed, failed, refunded)
        - metadata (JSONB)
        - error_message (TEXT)
        - created_at, completed_at (TIMESTAMPTZ)
      </payments>

      <push_tokens>
        - id (UUID, PK)
        - user_id (UUID, FK users)
        - token (TEXT)
        - platform (VARCHAR) - ios, android
        - active (BOOLEAN)
        - created_at, updated_at (TIMESTAMPTZ)
      </push_tokens>

      <notifications>
        - id (UUID, PK)
        - user_id (UUID, FK users)
        - title (VARCHAR)
        - body (TEXT)
        - data (JSONB)
        - read (BOOLEAN)
        - created_at (TIMESTAMPTZ)
      </notifications>

      <refresh_tokens>
        - id (UUID, PK)
        - user_id (UUID, FK users)
        - token_hash (VARCHAR)
        - expires_at (TIMESTAMPTZ)
        - revoked (BOOLEAN)
        - created_at (TIMESTAMPTZ)
      </refresh_tokens>
    </tables>
